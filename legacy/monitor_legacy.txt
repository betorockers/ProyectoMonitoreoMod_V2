class IPMonitor(customtkinter.CTkFrame):
    def __init__(self, master, ip, label, desconexiones_count=0): 
        super().__init__(master, corner_radius=10, fg_color="#2B2B2B",
                         border_width=2, border_color="#555555")

        self.ip = ip
        self.label = label
        self.status = "Verificando..."
        self.previous_status = None
        
        self.desconexiones_count = desconexiones_count 

        self.grid_columnconfigure(0, weight=1)

        self.label_name = customtkinter.CTkLabel(self, text=self.label, font=("Arial", 14, "bold"), text_color="#FFFFFF")
        self.label_name.grid(row=0, column=0, padx=10, pady=(10, 0))

        self.ip_label = customtkinter.CTkLabel(self, text=self.ip, font=("Arial", 12), text_color="#999999")
        self.ip_label.grid(row=1, column=0, padx=10, pady=(0, 5))

        self.status_icon_label = customtkinter.CTkLabel(self, text="❔", font=("Arial", 24), text_color="gray")
        self.status_icon_label.grid(row=2, column=0, padx=10, pady=(5, 0))

        self.status_text_label = customtkinter.CTkLabel(self, text=self.status, font=("Arial", 12), text_color="gray")
        self.status_text_label.grid(row=3, column=0, padx=10, pady=(0, 10))

    def update_status(self, new_status):
        self.previous_status = self.status
        self.status = new_status
        self.update_visuals()
        self.check_for_alert()

    def update_visuals(self):
        if self.status == "Conectado":
            self.configure(border_color="green")
            self.status_icon_label.configure(text="👍", text_color="green")
            self.status_text_label.configure(text="Conectado", text_color="green")
        elif self.status == "Desconectado":
            self.configure(border_color="red")
            self.status_icon_label.configure(text="👎", text_color="red")
            self.status_text_label.configure(text="Desconectado", text_color="red")
        else:
            self.configure(border_color="#555555")
            self.status_icon_label.configure(text="⏳", text_color="gray")
            self.status_text_label.configure(text=self.status, text_color="gray")
            
    def check_for_alert(self):
        if self.previous_status is not None and self.previous_status != self.status:
            if self.status == "Desconectado":
                self.send_alert(f"¡Alerta! {self.label} está desconectado.", "red")
                self.desconexiones_count += 1
            elif self.status == "Conectado":
                self.send_alert(f"¡Recuperado! {self.label} está en línea de nuevo.", "green")

    def send_alert(self, message, color):
        try:
            notification.notify(
                title=f"Estado del Equipo: {self.label}",
                message=message,
                app_name="Monitor de IPs",
                timeout=10
            )
        except Exception as e:
            print(f"Error al enviar la notificación: {e}")

        try:
            sound = pygame.mixer.Sound("assets/alerta.mp3")
            sound.play()
        except Exception as e:
            print(f"Error al reproducir el sonido: {e}")

    def create_monitors(self):
        print("Creando los monitores...")
        for widget in self.monitor_frame.winfo_children():
            widget.destroy()

        self.monitors = {}
        row = 0
        col = 0
        max_cols = 3

        for i in range(max_cols):
            self.monitor_frame.grid_columnconfigure(i, weight=1)

        for equipo in self.equipos_a_monitorear:
            desconexiones = equipo.get("desconexiones_count", 0) 
            monitor = IPMonitor(self.monitor_frame, equipo["ip"], equipo["label"], desconexiones_count=desconexiones)
            monitor.grid(row=row, column=col, padx=10, pady=10, sticky="nsew")
            
            self.monitors[equipo["ip"]] = monitor
            
            thread = threading.Thread(target=ping_ip, args=(equipo["ip"], monitor), daemon=True)
            thread.start()

            col += 1
            if col >= max_cols:
                col = 0
                row += 1

    def iniciar_refresco_automatico(self):
        self.after(120000, self.refrescar_y_programar)

    def refrescar_y_programar(self):
        print("Refrescando la información...")
        self.create_monitors()
        self.after(120000, self.refrescar_y_programar)

    def agregar_equipo(self):
        ip = self.ip_entry.get()
        etiqueta = self.etiqueta_entry.get()

        if ip and etiqueta:
            nuevo_equipo = {"ip": ip, "label": etiqueta, "desconexiones_count": 0} 
            self.equipos_a_monitorear.append(nuevo_equipo)
            self.create_monitors()
            self.ip_entry.delete(0, 'end')
            self.etiqueta_entry.delete(0, 'end')
        else:
            print("Error: La IP y la Etiqueta no pueden estar vacías.")
            
    def remover_equipo(self):
        ip_a_remover = self.remove_entry.get()
        
        if ip_a_remover:
            nueva_lista = [equipo for equipo in self.equipos_a_monitorear if equipo["ip"] != ip_a_remover]
            
            if len(nueva_lista) < len(self.equipos_a_monitorear):
                self.equipos_a_monitorear = nueva_lista
                self.create_monitors()
                print(f"Equipo con IP {ip_a_remover} eliminado.")
                self.remove_entry.delete(0, 'end')
            else:
                print(f"Error: No se encontró ningún equipo con la IP {ip_a_remover}.")
        else:
            print("Error: El campo de IP a eliminar no puede estar vacío.")

    def guardar_equipos(self):
        try:
            lista_para_guardar = []
            for equipo in self.equipos_a_monitorear:
                ip = equipo["ip"]
                if ip in self.monitors:
                    monitor = self.monitors[ip]
                    equipo["desconexiones_count"] = monitor.desconexiones_count
                lista_para_guardar.append(equipo)
            
            with open("equipos_guardados.json", "w") as file:
                json.dump(lista_para_guardar, file, indent=4)
            print("Configuración guardada exitosamente.")
        except Exception as e:
            print(f"Error al guardar la configuración: {e}")

    def cargar_equipos(self):
        try:
            with open("equipos_guardados.json", "r") as file:
                self.equipos_a_monitorear = json.load(file)
            print("Configuración cargada exitosamente.")
            self.create_monitors()
        except FileNotFoundError:
            print("No se encontró el archivo 'equipos_guardados.json'.")
        except Exception as e:
            print(f"Error al cargar la configuración: {e}")
            
    def generar_reporte(self):
        report_filename = f"reporte_desconexiones_{datetime.datetime.now().strftime('%Y-%m-%d_%H-%M-%S')}.txt"
        
        try:
            with open(report_filename, "w") as file:
                file.write("--- Reporte de Desconexiones de Equipos ---\n")
                file.write(f"Fecha y Hora: {datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n\n")
                
                for equipo in self.equipos_a_monitorear:
                    ip = equipo["ip"]
                    label = equipo["label"]
                    monitor = self.monitors.get(ip)
                    if monitor:
                        count = monitor.desconexiones_count
                        file.write(f"Equipo: {label} ({ip})\n")
                        file.write(f"  Número de Desconexiones: {count}\n")
                        file.write("-" * 30 + "\n")
                        
            print(f"Reporte generado exitosamente en {report_filename}")
        except Exception as e:
            print(f"Error al generar el reporte: {e}")
